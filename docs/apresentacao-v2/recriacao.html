SERVICES/REGRASASSENTO.TS
export function estaDisponivel(
  a: AssentoDaSessao, agora: Date
): boolean {
  if (a.status === "LIVRE") return true;
  if (a.status === "VENDIDO") return false;
  if (a.reservaStatus === "EXPIRADA"
   || a.reservaStatus === "CANCELADA") return true;
  // reservado sem dados visíveis: ocupado
  if (a.reservaStatus !== "PENDENTE"
   || a.reservaExpiraEm === null) return false;
  // reserva vencida já conta como livre
  return a.reservaExpiraEm.getTime() <= agora.getTime();
}

O TESTE QUE PEGOU UM BUG
Travar e ler não é a mesma coisa.
Antes · 20 de 20 aceitas
Depois · 1 aceita, 19 recusadas
SELECT a.*, r.status
  FROM assentos_sessao a
  LEFT JOIN reservas r ON ...
  FOR UPDATE OF a;   -- um comando só
SELECT ... FOR UPDATE;   -- 1. trava
SELECT a.*, r.status       -- 2. lê
  FROM assentos_sessao a
  LEFT JOIN reservas r ON ...;
Em READ COMMITTED, a linha travada volta atualizada, mas o
JOIN ainda lê reservas no snapshot antigo.
Dois comandos: o segundo enxerga o que já foi confirmado. E
reservado sem dados visíveis conta como ocupado.
